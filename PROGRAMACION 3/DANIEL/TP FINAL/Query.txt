-- Tablas

CREATE TABLE Usuarios (
    id_usuario INT PRIMARY KEY IDENTITY(1,1),
    nombre VARCHAR(100) NOT NULL,
    rol VARCHAR(20) NOT NULL -- 'Admin' o 'Usuario'
);

CREATE TABLE Apuestas (
    id_apuesta INT PRIMARY KEY IDENTITY(1,1),
    enunciado VARCHAR(255) NOT NULL,
    opcion_a VARCHAR(100) NOT NULL, -- Ej: 'Gana Sinner'
    opcion_b VARCHAR(100) NOT NULL, -- Ej: 'Gana Alcaraz'
    estado VARCHAR(20) DEFAULT 'Abierta' 
);

CREATE TABLE Jugadas (
    id_jugada INT PRIMARY KEY IDENTITY(1,1),
    id_usuario INT NOT NULL,
    id_apuesta INT NOT NULL,
    eleccion CHAR(1) NOT NULL CHECK (eleccion IN ('A', 'B')), -- Solo permite 'A' o 'B'
    monto DECIMAL(12, 2) NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES Usuarios(id_usuario),
    FOREIGN KEY (id_apuesta) REFERENCES Apuestas(id_apuesta)
);


-- Pedidos

SELECT COUNT(DISTINCT id_usuario) AS total_apostadores 
FROM Jugadas 
WHERE id_apuesta = 1; -- Reemplazar 1 por el ID de la apuesta

SELECT 
    eleccion, 
    COUNT(id_jugada) AS cantidad_jugadas, 
    SUM(monto) AS dinero_acumulado 
FROM Jugadas 
WHERE id_apuesta = 1 -- Reemplazar 1 por el ID de la apuesta
GROUP BY eleccion;
